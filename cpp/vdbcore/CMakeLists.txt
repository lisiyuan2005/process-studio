cmake_minimum_required(VERSION 3.18)
project(vdbcore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Python 3.12 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE nanobind_ROOT)
find_package(nanobind CONFIG REQUIRED)
find_library(OPENVDB_LIB openvdb HINTS $ENV{CONDA_PREFIX}/lib REQUIRED)
find_package(TBB REQUIRED)
nanobind_add_module(_vdbcore NB_STATIC vdbcore.cpp)
target_include_directories(_vdbcore PRIVATE $ENV{CONDA_PREFIX}/include)
target_link_libraries(_vdbcore PRIVATE ${OPENVDB_LIB} TBB::tbb)
target_compile_options(_vdbcore PRIVATE -O3)
