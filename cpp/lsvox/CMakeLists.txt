cmake_minimum_required(VERSION 3.18)
project(lsvox LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Python 3.12 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE nanobind_ROOT)
find_package(nanobind CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
set(TP ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party)
nanobind_add_module(_lsvox NB_STATIC lsvox.cpp)
target_include_directories(_lsvox PRIVATE ${TP}/ViennaLS/include/viennals ${TP}/ViennaHRLE/include/viennahrle ${TP}/ViennaCore/include/viennacore)
target_link_libraries(_lsvox PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(_lsvox PRIVATE -O3)
